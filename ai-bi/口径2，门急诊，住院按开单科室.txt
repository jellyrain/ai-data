--一
--门急诊：科室+病种维度的就诊费用汇总（按开单科室计算）
select 
left(a.pdate,7)  as 年月
,a.pat_kind_name as 就诊类型
,a.deptname      as 开单科室
,b.icdecode      as 病种编码
,b.icdename      as 病种名称
--,sum(cast(b.pvalue_rc as float))     as 就诊人次 
,sum(a.pvalue)        as 总费用 
from dbo.ai_pat_fee_detail_mz2 a (nolock)
inner join dbo.ai_pat_info_mz b (nolock) on b.visit_no = a.visit_no
where a.pdate between '2025-05-01' and '2025-05-31 23:59:59' 
and a.pname in ('门诊收入','急诊收入')
and a.visit_no !='0'
and isnull(a.visit_no,'') !=''
and b.visit_no !='0'
and isnull(b.visit_no,'') !=''
group by left(a.pdate,7),a.pat_kind_name,a.deptname,b.icdecode,b.icdename
--住院：科室+病种维度的就诊费用汇总（按开单科室计算）

and a.visit_no !='0'
and isnull(a.visit_no,'') !=''
and b.visit_no !='0'

group by left(a.pdate,7),a.pat_kind_name,a.deptname,b.icdecode,b.icdename

from dbo.ai_pat_fee_detail_mz2 a (nolock)

and a.visit_no !='0'
and isnull(a.visit_no,'') !=''
and b.visit_no !='0'
and isnull(b.visit_no,'') !=''

group by left(a.pdate,7),a.pat_kind_name,a.deptname,b.icdecode,b.icdename,a.pname
